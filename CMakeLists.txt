cmake_minimum_required(VERSION 3.25)
project(KinokoDS VERSION 0.1.0 LANGUAGES CXX)

# CMake 3.29 does not yet model GCC's C++26 mode. Require the compiler
# flag explicitly instead of silently falling back to an older standard.
include(CheckCXXCompilerFlag)
check_cxx_compiler_flag("-std=c++26" KINOKO_HAS_CXX26)
if(NOT KINOKO_HAS_CXX26)
  message(FATAL_ERROR "A compiler accepting -std=c++26 is required")
endif()
add_library(kinoko_options INTERFACE)
target_compile_options(kinoko_options INTERFACE -std=c++26 -Wall -Wextra -Wpedantic)

add_library(kinoko_sim src/sim/world.cpp src/sim/longitudinal.cpp src/sim/movement.cpp)
target_include_directories(kinoko_sim PUBLIC include)
target_link_libraries(kinoko_sim PUBLIC kinoko_options)
add_library(kinoko_assets src/assets/parameters.cpp src/assets/selection.cpp src/assets/preparation.cpp)
target_include_directories(kinoko_assets PUBLIC include)
target_link_libraries(kinoko_assets PUBLIC kinoko_options)
add_executable(kinoko_parameters tools/parameters.cpp)
target_link_libraries(kinoko_parameters PRIVATE kinoko_assets)
add_executable(kinoko_verify_surface tools/verify_surface.cpp)
target_link_libraries(kinoko_verify_surface PRIVATE kinoko_assets)
add_executable(kinoko_sync_dump tools/sync_dump.cpp)
target_link_libraries(kinoko_sync_dump PRIVATE kinoko_assets)
add_executable(kinoko_offline_runner tools/offline_runner.cpp)
target_link_libraries(kinoko_offline_runner PRIVATE kinoko_assets kinoko_sim)
add_executable(kinoko_movement_runner tools/movement_runner.cpp)
target_link_libraries(kinoko_movement_runner PRIVATE kinoko_sim)
add_library(kinoko_trace src/replay/trace.cpp)
target_include_directories(kinoko_trace PUBLIC include)
target_link_libraries(kinoko_trace PUBLIC kinoko_options)
add_executable(kinoko_compare tools/compare.cpp)
target_link_libraries(kinoko_compare PRIVATE kinoko_trace)
add_executable(kinoko_accel tools/accel.cpp)
target_link_libraries(kinoko_accel PRIVATE kinoko_sim kinoko_trace)

include(CTest)
if(BUILD_TESTING)
  add_executable(kinoko_longitudinal_tests tests/longitudinal.cpp)
  target_link_libraries(kinoko_longitudinal_tests PRIVATE kinoko_assets kinoko_sim)
  add_test(NAME longitudinal COMMAND kinoko_longitudinal_tests)
  add_executable(kinoko_selection_tests tests/selection.cpp)
  target_link_libraries(kinoko_selection_tests PRIVATE kinoko_assets)
  add_test(NAME parameter_selection COMMAND kinoko_selection_tests)
  add_executable(kinoko_surface_tests tests/surface_factor.cpp)
  target_link_libraries(kinoko_surface_tests PRIVATE kinoko_assets)
  add_test(NAME surface_factor COMMAND kinoko_surface_tests)
  add_executable(kinoko_parameter_tests tests/parameters.cpp)
  target_link_libraries(kinoko_parameter_tests PRIVATE kinoko_assets)
  add_test(NAME parameters COMMAND kinoko_parameter_tests)
  add_executable(kinoko_tests tests/tests.cpp)
  target_link_libraries(kinoko_tests PRIVATE kinoko_sim kinoko_trace)
  add_test(NAME unit COMMAND kinoko_tests)
  add_test(NAME identical_trace COMMAND kinoko_compare
    ${PROJECT_SOURCE_DIR}/tests/fixtures/synthetic.csv
    ${PROJECT_SOURCE_DIR}/tests/fixtures/synthetic.csv)
  add_test(NAME replay_acceleration COMMAND kinoko_accel
    ${PROJECT_SOURCE_DIR}/tests/fixtures/synthetic.csv
    ${PROJECT_BINARY_DIR}/replayed.csv)
  set_tests_properties(replay_acceleration PROPERTIES FIXTURES_SETUP replay)
  add_test(NAME compare_acceleration COMMAND kinoko_compare
    ${PROJECT_SOURCE_DIR}/tests/fixtures/synthetic.csv
    ${PROJECT_BINARY_DIR}/replayed.csv)
  set_tests_properties(compare_acceleration PROPERTIES FIXTURES_REQUIRED replay)
endif()
